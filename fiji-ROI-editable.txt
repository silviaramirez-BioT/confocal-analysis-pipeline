function processFolder(dir) {
    list = getFileList(dir);
    for (k = 0; k < list.length; k++) {
        fName = list[k];
        if (File.isDirectory(dir + fName)) {
            processFolder(dir + fName);
        } else if (endsWith(toLowerCase(fName), ".lif") || endsWith(toLowerCase(fName), ".czi")) {
            processLIF(dir, fName);
        } else if (endsWith(toLowerCase(fName), "azul.tif") || endsWith(toLowerCase(fName), "blue.tif")) {
            processTIF(dir, fName);
        }
    }
}

function processLIF(dir, filename) {
    path = dir + filename;
    Ext.setId(path);
    Ext.getSeriesCount(numSeries);
    for (i = 0; i < numSeries; i++) {
        run("Bio-Formats Importer", "open=[" + path + "] autoscale color_mode=Composite view=Hyperstack stack_order=XYCZT series_" + (i+1));
        analyzeSafe(getTitle());
    }
}

function processTIF(dir, blueFile) {
    open(dir + blueFile); rename("C1");
    greenFile = replace(blueFile, "azul", "verde"); greenFile = replace(greenFile, "blue", "green");
    redFile = replace(blueFile, "azul", "rojo");    redFile = replace(redFile, "blue", "red");
    
    if (!File.exists(dir+greenFile)) greenFile=replace(greenFile, ".tif", ".TIF");
    
    if (File.exists(dir + greenFile)) { open(dir + greenFile); rename("C2"); }
    if (File.exists(dir + redFile))   { open(dir + redFile);   rename("C3"); }
    
    strMerge = "c3=C1 ";
    if (isOpen("C2")) strMerge = strMerge + "c2=C2 ";
    if (isOpen("C3")) strMerge = strMerge + "c1=C3 ";
    
    run("Merge Channels...", strMerge + "create keep");
    titulo = replace(blueFile, " - azul.tif", "");
    rename(titulo);
    analyzeSafe(titulo);
    run("Close All");
}

function analyzeSafe(titulo) {
    safeTitle = replace(titulo, ".lif", "");
    safeTitle = replace(safeTitle, " - ", "_");
    safeTitle = replace(safeTitle, " ", "_");
    
    getDimensions(width, height, channels, slices, frames);
    
    if (channels > 1) {
        run("Split Channels");
    } else {
        rename("C1-Temp");
    }
    
    titulosAbiertos = getList("image.titles");
    for (j=0; j<titulosAbiertos.length; j++) {
        t = titulosAbiertos[j];
        selectWindow(t);
        if (indexOf(t, "C1") >= 0) rename("CANAL_1");
        else if (indexOf(t, "C2") >= 0) rename("CANAL_2");
        else if (indexOf(t, "C3") >= 0) rename("CANAL_3");
    }
    
    winBlue = "CANAL_1"; 
    winGreen = "CANAL_2"; 
    winRed = "CANAL_3";
    
    hasBlue = isOpen(winBlue);
    hasGreen = isOpen(winGreen);
    hasRed = isOpen(winRed);
    
    if (!hasBlue && hasRed && !hasGreen) {
        selectWindow(winRed); rename("CANAL_1");
        winBlue = "CANAL_1"; hasBlue = true; hasRed = false;
    }

    if (hasBlue) { selectWindow(winBlue); saveAs("Tiff", dirCanales + "Azul_" + safeTitle); rename(winBlue); }
    if (hasGreen) { selectWindow(winGreen); saveAs("Tiff", dirCanales + "Verde_" + safeTitle); rename(winGreen); }
    if (hasRed) { selectWindow(winRed); saveAs("Tiff", dirCanales + "Rojo_" + safeTitle); rename(winRed); }

    if (hasBlue) {
        selectWindow(winBlue);
        run("Duplicate...", "title=Suma_Temp");
        
        if (hasGreen) { imageCalculator("Add", "Suma_Temp", winGreen); }
        if (hasRed)   { imageCalculator("Add", "Suma_Temp", winRed); }
        
        selectWindow("Suma_Temp");
        run("Subtract Background...", "rolling=50");
        run("Gaussian Blur...", "sigma=2");
        setAutoThreshold("Triangle dark");
        run("Convert to Mask");
        run("Fill Holes");
        run("Watershed");
        roiManager("Reset");
        run("Analyze Particles...", "size=0-Infinity exclude add");
        close("Suma_Temp");
    }

    strMerge = "";
    if (hasRed) strMerge = strMerge + "c1=["+winRed+"] ";
    if (hasGreen) strMerge = strMerge + "c2=["+winGreen+"] ";
    if (hasBlue) strMerge = strMerge + "c3=["+winBlue+"] ";
    
    if (strMerge != "") { 
        run("Merge Channels...", strMerge + "create keep"); 
        rename("REF_VISUAL"); 
    } else if (hasBlue) { 
        selectWindow(winBlue); 
    }
    
    setTool("freehand");
    roiManager("Show All");
    waitForUser("EDICIÓN", "Edita ROIs y dale OK.\n(No cierres las ventanas manualmente)");
    
    if (isOpen("REF_VISUAL")) close("REF_VISUAL");

    newImage("Mascara_Final", "8-bit black", width, height, 1);
    if (roiManager("count") > 0) { roiManager("Deselect"); roiManager("Fill"); }
    saveAs("Tiff", dirMascaras + "Mascara_" + safeTitle);
    close();

    count = roiManager("count");
    if (count > 0) {
        for (r = 0; r < count; r++) {
            roiManager("Select", r);
            
            meanB=0; area=0;
            if (hasBlue) {
                selectWindow(winBlue); resetThreshold(); 
                run("Measure");
                area=getResult("Area",nResults-1); meanB=getResult("Mean",nResults-1);
            }
            
            meanG=0; intDenG=0; percG=0;
            if (hasGreen) {
                selectWindow(winGreen); resetThreshold();
                run("Measure");
                meanG=getResult("Mean",nResults-1); intDenG=getResult("IntDen",nResults-1);
                
                run("Duplicate...", "title=TG"); setAutoThreshold("Moments dark"); run("Measure"); 
                percG=getResult("%Area",nResults-1); close("TG");
            }
            
            meanR=0; intDenR=0; percR=0;
            if (hasRed) {
                selectWindow(winRed); resetThreshold();
                run("Measure");
                meanR=getResult("Mean",nResults-1); intDenR=getResult("IntDen",nResults-1);
                
                run("Duplicate...", "title=TR"); setAutoThreshold("Moments dark"); run("Measure"); 
                percR=getResult("%Area",nResults-1); close("TR");
            }
            
            print(f, titulo + "," + (r+1) + "," + area + "," + meanB + "," + meanG + "," + meanR + "," + intDenG + "," + intDenR + "," + percG + "," + percR);
        }
        
        if (strMerge != "") {
            run("Merge Channels...", strMerge + "create keep");
            run("RGB Color");
            saveAs("Jpeg", dirMerged + "Merged_" + safeTitle + ".jpg");
            roiManager("Show All without labels"); run("Flatten");
            saveAs("Jpeg", dirOutlines + "Check_" + safeTitle + ".jpg");
        }
    }
    roiManager("Delete");
    close("*");
}
